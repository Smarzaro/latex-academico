\begin{lstlisting}[language=Python,
    caption={Função em Python, com numeração de linhas e sintaxe colorida},
    label={cod:python}]
def media(valores):
    """Calcula a média aritmética de uma lista."""
    if not valores:
        return 0.0
    total = 0  # acumulador
    for v in valores:
        total += v
    return total / len(valores)

print(f"Média: {media([8.5, 7.0, 9.25]):.2f}")
\end{lstlisting}
