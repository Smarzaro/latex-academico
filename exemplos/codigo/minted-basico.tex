\begin{listing}[htbp]
  \caption{Função em Python, colorida com o minted}\label{lst:minted}
\begin{minted}{python}
def media(valores):
    """Calcula a média aritmética de uma lista."""
    if not valores:
        return 0.0
    return sum(valores) / len(valores)
\end{minted}
