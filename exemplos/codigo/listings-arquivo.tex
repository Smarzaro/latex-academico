% Exemplo: listar parte de um arquivo (lstinputlisting com firstline/lastline), sem copiar o código
% Compilar: latexmk -pdf listings-arquivo.tex     Testado com: abntex2, TeX Live 2024 ou mais recente
\documentclass[12pt, a4paper, brazilian]{abntex2}
\usepackage[T1]{fontenc}
\input{estilo-codigo}

\begin{document}

O Código~\ref{lst:arquivo} lista um trecho do arquivo \texttt{exemplo.py}.

%%BEGIN:listingsarquivo
\lstinputlisting[language=Python, firstline=4, lastline=11, numbers=none,
    caption={Trecho de um arquivo (linhas 4 a 11), sem numeração de linhas},
    label={lst:arquivo}]{assets/exemplo.py}
\fonte{Arquivo \texttt{exemplo.py}, do autor.}
%%END:listingsarquivo

\end{document}
