%%BEGIN:gobble
\begin{itemize}
  \item Com \texttt{gobble=4}, os 4 primeiros caracteres de cada linha são descartados:
    \begin{lstlisting}[language=Python, gobble=4]
    for v in valores:
        total += v
    \end{lstlisting}
  \item Com \texttt{autogobble}, o recuo é medido na primeira linha:
    \begin{lstlisting}[language=Python, autogobble]
        for v in valores:
            total += v
    \end{lstlisting}
\end{itemize}
%%END:gobble
