\begin{lstlisting}[language=SQL, style=codigopb,
    caption={O mesmo código em preto e branco (estilo \texttt{codigopb})},
    label={lst:pb}]
-- Alunos aprovados por disciplina
SELECT d.nome, COUNT(*) AS aprovados
FROM matricula m
JOIN disciplina d ON d.id = m.disciplina_id
WHERE m.nota >= 6.0 AND m.situacao = 'APROVADO'
GROUP BY d.nome
ORDER BY aprovados DESC;
\end{lstlisting}
