from statistics import mean

def situacao(notas, minimo=6.0):
    """Retorna a media e a situacao."""
    m = mean(notas) if notas else 0  # media
    if m >= minimo:
        return m, "Aprovado"
    return m, "Reprovado"

print(situacao([7.5, 5.0, 8.25]))
\end{lstlisting}
