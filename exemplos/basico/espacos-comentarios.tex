\documentclass[a4paper]{article}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage[brazilian]{babel}

\begin{document}

%%BEGIN:espacos
Esses   espaços   em    branco  extras    serão ignorados.

linha são tratadas como
espaço.

%%END:espacos

\clearpage

%%BEGIN:comentarios

% espalhado por
% várias linhas.

%%END:comentarios

\end{document}
