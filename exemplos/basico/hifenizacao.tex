% Exemplo: indicar a hifenização de palavras desconhecidas (\- no texto ou \hyphenation)
% Compilar: latexmk -pdf hifenizacao.tex           Testado com: TeX Live 2024 ou mais recente
% Cada trecho fica em uma página. Normalmente \hyphenation vai no preâmbulo; aqui ele aparece no
% meio do texto só para que a primeira página mostre o problema.
\documentclass[a4paper]{article}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage[brazilian]{babel}

\begin{document}

%%BEGIN:hifenproblema
\begin{minipage}{3.5cm}
Estudos da Microssoftware apontam a instagramização da humanidade e a
tiktokização das relações.
\end{minipage}
%%END:hifenproblema

\clearpage

%%BEGIN:hifenmanual
\begin{minipage}{3.5cm}
Estudos da Micros\-soft\-ware apontam a ins\-ta\-gra\-mi\-za\-ção da humanidade
e a tik\-to\-ki\-za\-ção das relações.
\end{minipage}
%%END:hifenmanual

\clearpage

%%BEGIN:hifencomando
\hyphenation{Micros-soft-ware ins-ta-gra-mi-za-ção tik-to-ki-za-ção}

\begin{minipage}{3.5cm}
Estudos da Microssoftware apontam a instagramização da humanidade e a
tiktokização das relações.
\end{minipage}
%%END:hifencomando

\end{document}
