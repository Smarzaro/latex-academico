% Exemplo: referências que escrevem o tipo do elemento: \autoref (hyperref) e \cref (cleveref)
% Compilar: latexmk -pdf autoref-cleveref.tex      Testado com: TeX Live 2024 ou mais recente
\documentclass[a4paper]{article}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage[brazilian]{babel}
%%BEGIN:cleverefpreambulo
\usepackage[hidelinks]{hyperref}   % links e \autoref
\usepackage[brazilian]{cleveref}   % depois do hyperref; nomes e conjunções em português
% só para elementos que o pacote não conhece (ex.: um flutuante "quadro"),
% ou para trocar um nome: \crefname{tipo}{singular}{plural}
\crefname{quadro}{quadro}{quadros}
\Crefname{quadro}{Quadro}{Quadros}
%%END:cleverefpreambulo

\begin{document}

%%BEGIN:cleverefuso
\section{Introdução}\label{sec:intro}
Com \verb|\autoref|: veja a \autoref{sec:metodo}. Com \verb|\cref|: veja a
\cref{sec:metodo}; no início da frase, \verb|\Cref|: \Cref{sec:metodo}
descreve o método. Vários rótulos de uma vez: \cref{sec:intro,sec:metodo};
uma tabela: \cref{tab:dados}.

\begin{table}[h]
  \centering
  \caption{Dados}\label{tab:dados}
  \begin{tabular}{cc} 1 & 2 \\ \end{tabular}
\end{table}

\section{Método}\label{sec:metodo}
%%END:cleverefuso

\end{document}
