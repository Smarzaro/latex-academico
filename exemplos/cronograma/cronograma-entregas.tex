% Exemplo: cronograma com períodos de execução e marcos de entrega em cores (xcolor + colortbl)
% Para cronogramas longos, que passam de uma página, troque o tabular por um longtable (pacote
% longtable), com o cabeçalho repetido por \endhead.
% Compilar: latexmk -pdf cronograma-entregas.tex   Testado com: abntex2, xcolor, booktabs, TeX Live 2024+
\documentclass[12pt, a4paper, brazilian]{abntex2}
\usepackage[T1]{fontenc}
\usepackage{array}
\usepackage{booktabs}
\usepackage[table]{xcolor}            % \cellcolor (carrega o colortbl)
\usepackage{amssymb}                  % \blacklozenge

%%BEGIN:entregaspreambulo
\definecolor{periodo}{RGB}{158,202,225}  % meses de execução
\definecolor{entrega}{RGB}{8,81,156}     % meses com entrega (marco)
\newcommand{\mes}{\cellcolor{periodo}}
\newcommand{\marco}{\cellcolor{entrega}%
  \textcolor{white}{\scriptsize$\blacklozenge$}}
\newfloat{quadro}{loq}{Quadro}           % a abntex2 não define o float Quadro
\newlistof{listofquadros}{loq}{Lista de quadros}
\newlistentry{quadro}{loq}{0}         % linha da lista, com pontilhado e página
\newcommand{\quadroautorefname}{Quadro}  % nome usado pelo \autoref
%%END:entregaspreambulo

\begin{document}

O \autoref{qua:entregas} destaca as entregas previstas no projeto.
