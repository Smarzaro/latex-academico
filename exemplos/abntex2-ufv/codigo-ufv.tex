%%BEGIN:codigoufv
\begin{codigo}[htbp]
\begin{ilustracao}[\linewidth]{\caption{Função em Python}\label{cod:ufv}}%
                  {\elaboracaopropria}
\begin{lstlisting}[language=Python]
def media(valores):
    """Calcula a média aritmética de uma lista."""
    if not valores:
        return 0.0
    return sum(valores) / len(valores)
\end{lstlisting}
\end{ilustracao}
\end{codigo}
